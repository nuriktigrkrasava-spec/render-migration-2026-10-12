# Карта желаний и планы — план реализации

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Локальная страница с картой желаний и планами на 3/6/12 месяцев, где каждая цель раскрывается в дерево задач.

**Architecture:** Статическая страница без сборки, открывается двойным кликом по `index.html`. Чистая логика вынесена в `js/model.js` и покрыта тестами через `node --test`; хранение изолировано в `js/store.js`; рендер и события — в `js/ui.js`. Данные — один JSON, живут в памяти браузера с копией на диск.

**Tech Stack:** HTML, CSS, ванильный JavaScript (классические скрипты, без сборщика), `node:test` для тестов. Никаких зависимостей.

**Spec:** `docs/superpowers/specs/2026-08-22-wish-map-design.md`

## Global Constraints

Действуют во всех задачах без исключения.

- Страница открывается **двойным кликом** по `index.html`, протокол `file://`.
- **Никаких ES-модулей.** `<script type="module">` на `file://` блокируется CORS и страница молча не запустится. Только классические `<script src="...">`.
- **Никаких внешних загрузок:** ни CDN, ни Google Fonts, ни шрифтов с хостов. Работает без интернета. Шрифты — системные (`-apple-system`, `BlinkMacSystemFont`, `SF Pro`), это же требует и `apple-design`.
- Горизонты — только числа `3`, `6`, `12`. Других не бывает.
- Вложенность задач — ровно два уровня: задача и подзадача. Глубже не создавать.
- У каждой цели есть стартовая точка: `startedAt` (дата, ставится сама, не
  редактируется) и `start` (текст, может быть пустым). У задач её нет.
- Счётчик выполнения считает **прямых детей**, а не листья дерева.
- Ключ хранилища — `wishmap-v1`. Поле `version` в данных — `1`.
- Картинки — в папке `images/`, в данных хранится только имя файла.
- Фото присылает владелец, кладёт и привязывает автор проекта. Интерфейса загрузки нет.
- Палитра — **очень богатые, глубокие оттенки**. Насыщенные драгоценные тона, тональные слои. Никакой бледности, пастели и плоской заливки.
- Визуальная основа — скилл `apple-design`, палитра подбирается через `ui-ux-pro-max`. Не придумывать на глазок.
- `prefers-reduced-motion` отключает всё движение.
- Тесты запускаются `node --test` из корня проекта. Зависимости не ставить.

## Структура файлов

| Файл | За что отвечает |
|---|---|
| `index.html` | Разметка, подключение стилей и скриптов |
| `styles.css` | Всё оформление |
| `js/model.js` | Чистая логика: дерево, счётчики, операции, разбор JSON. Не трогает DOM |
| `js/store.js` | Хранение: память браузера, скачивание и чтение файла |
| `js/ui.js` | Рендер и обработчики событий |
| `tests/model.test.js` | Тесты логики |
| `tests/store.test.js` | Тесты разбора и валидации |
| `images/` | Фотографии мечт |

Каждый JS-файл обёрнут так, чтобы работать и в браузере через `file://`, и в Node при тестах:

(function (root) {
  'use strict';

  const HORIZONS = [3, 6, 12];
  const VERSION = 1;

  const DEFAULT_DATA = {
    version: VERSION,
    updatedAt: null,
    areas: [
      { id: 'money', name: 'Деньги' },
      { id: 'health', name: 'Здоровье' },
      { id: 'skills', name: 'Навыки' },
      { id: 'people', name: 'Отношения' },
      { id: 'home', name: 'Дом' }
    ],
    dreams: [],
    goals: []
  };

  let counter = 0;
  function createId(prefix) {
    counter += 1;
    return prefix + Date.now().toString(36) + counter.toString(36);
  }

  function validate(data) {
    if (!data || typeof data !== 'object') return { ok: false, error: 'Данные не объект' };
    if (data.version !== VERSION) return { ok: false, error: 'Другая версия данных' };
    if (!Array.isArray(data.areas)) return { ok: false, error: 'Нет списка сфер' };
    if (!Array.isArray(data.dreams)) return { ok: false, error: 'Нет списка мечт' };
    if (!Array.isArray(data.goals)) return { ok: false, error: 'Нет списка целей' };

    const areaIds = new Set(data.areas.map(a => a.id));
    for (const goal of data.goals) {
      if (!HORIZONS.includes(goal.horizon)) {
        return { ok: false, error: 'Недопустимый горизонт у цели: ' + goal.title };
      }
      if (!areaIds.has(goal.areaId)) {
        return { ok: false, error: 'Цель ссылается на несуществующую сферу: ' + goal.title };
      }
      if (typeof goal.startedAt !== 'string' || !/^\d{4}-\d{2}-\d{2}$/.test(goal.startedAt)) {
        return { ok: false, error: 'У цели нет даты старта: ' + goal.title };
      }
    }
    return { ok: true, data: data };
  }

  function parse(json) {
    let raw;
    try {
      raw = JSON.parse(json);
    } catch (e) {
      return { ok: false, error: 'Не удалось разобрать файл' };
    }
    return validate(raw);
  }

  function serialize(data) {
    const copy = Object.assign({}, data, { updatedAt: new Date().toISOString() });
    return JSON.stringify(copy, null, 2);
  }

  const Model = { HORIZONS, VERSION, DEFAULT_DATA, createId, validate, parse, serialize };

  if (typeof module !== 'undefined' && module.exports) module.exports = Model;
  else root.Model = Model;
})(typeof globalThis !== 'undefined' ? globalThis : this);
```

  function children(node) {
    return Array.isArray(node.tasks) ? node.tasks : [];
  }

  function isDone(node) {
    const kids = children(node);
    if (kids.length === 0) return node.done === true;
    return kids.every(isDone);
  }

  function progress(node) {
    const kids = children(node);
    if (kids.length === 0) return null;
    return { done: kids.filter(isDone).length, total: kids.length };
  }

  function canToggle(node) {
    return children(node).length === 0;
  }

  function findNode(data, id) {
    function walk(nodes) {
      for (const node of nodes) {
        if (node.id === id) return node;
        const found = walk(children(node));
        if (found) return found;
      }
      return null;
    }
    return walk(data.goals);
  }

  function clone(data) {
    return JSON.parse(JSON.stringify(data));
  }

  function toggle(data, id) {
    const next = clone(data);
    const node = findNode(next, id);
    if (!node || !canToggle(node)) return next;
    node.done = !node.done;
    return next;
  }
```

    function walk(nodes, level) {
      for (const node of nodes) {
        if (node.id === id) return level;

        if (found !== -1) return found;

(function (root) {
  'use strict';

  const KEY = 'wishmap-v1';
  const Model = (typeof module !== 'undefined' && module.exports)
    ? require('./model.js')
    : root.Model;

  function available(storage) {
    try {
      storage.setItem(KEY + '-probe', '1');
      storage.removeItem(KEY + '-probe');
      return true;
    } catch (e) {
      return false;
    }
  }

  function load(storage) {
    let raw = null;
    try {
      raw = storage.getItem(KEY);
    } catch (e) {
      return {
        data: Model.clone(Model.DEFAULT_DATA),
        source: 'default',
        warning: 'Браузер не даёт сохранять на этой странице. Правки не переживут перезагрузку — пользуйтесь кнопкой «Сохранить копию».'
      };
    }
    if (!raw) {
      return { data: Model.clone(Model.DEFAULT_DATA), source: 'default', warning: null };
    }
    const res = Model.parse(raw);
    if (!res.ok) {
      return {
        data: Model.clone(Model.DEFAULT_DATA),
        source: 'default',
        warning: 'Сохранённые данные повреждены: ' + res.error + '. Открыт стартовый набор.'
      };
    }
    return { data: res.data, source: 'storage', warning: null };
  }

  function save(storage, data) {
    try {
      storage.setItem(KEY, Model.serialize(data));
      return true;
    } catch (e) {
      return false;
    }
  }

  function exportName(date) {
    const d = date || new Date();
    const iso = d.toISOString().slice(0, 10);
    return 'plans-' + iso + '.json';
  }

  const Store = { KEY, available, load, save, exportName };

  if (typeof module !== 'undefined' && module.exports) module.exports = Store;
  else root.Store = Store;
})(typeof globalThis !== 'undefined' ? globalThis : this);

  function editInPlace(node, current, onDone) {
    const input = document.createElement('input');
    input.type = 'text';
    input.className = 'inline-edit';
    input.value = current || '';
    node.replaceWith(input);
    input.focus();
    input.setSelectionRange(input.value.length, input.value.length);

    let settled = false;
    function finish(save) {
      if (settled) return;
      settled = true;
      const value = input.value;
      input.replaceWith(node);
      if (save) onDone(value);
      else renderAll();
    }
    input.addEventListener('keydown', e => {
      if (e.key === 'Enter') { e.preventDefault(); finish(true); }
      if (e.key === 'Escape') { e.preventDefault(); finish(false); }
    });
    input.addEventListener('blur', () => finish(true));
  }

  function addInPlace(container, placeholder, onDone) {
    const input = document.createElement('input');
    input.type = 'text';
